# A directory-scoped package probe: rejected imported targets stay in this
# directory and cannot poison the fallback source build in a sibling directory.
find_program(_zahlen_host_slangc NAMES slangc NO_CACHE)
if(NOT _zahlen_host_slangc)
    message(STATUS "Slang: no host slangc found")
    return()
endif()
if(CMAKE_CROSSCOMPILING)
    message(STATUS "Slang: host package skipped for cross compilation")
    return()
endif()
execute_process(COMMAND "${_zahlen_host_slangc}" -version
    RESULT_VARIABLE _rc OUTPUT_VARIABLE _out ERROR_VARIABLE _err TIMEOUT 15)
if(NOT "${_rc}" STREQUAL "0" OR NOT "${_out} ${_err}" MATCHES "([0-9][0-9][0-9][0-9]\\.[0-9]+)")
    message(STATUS "Slang: reject host compiler ${_zahlen_host_slangc}: version probe failed")
    return()
endif()
set(_compiler_version "${CMAKE_MATCH_1}")
if(NOT ZHLN_SLANG_ALLOW_KNOWN_BAD AND _compiler_version MATCHES "^(2025\\.24|2026\\.1)$")
    message(STATUS "Slang: reject host ${_compiler_version}: known-bad family (#9500)")
    return()
endif()

# Native package discovery, including CMAKE_PREFIX_PATH and slang_DIR.
# No SDK hints, prefix reconstruction, or handwritten library layout.
find_package(slang CONFIG QUIET)
if(NOT slang_FOUND)
    message(STATUS "Slang: no host slang config package found")
    return()
endif()
if(TARGET slang::slang)
    set(_target slang::slang)
elseif(TARGET slang)
    set(_target slang)
else()
    message(STATUS "Slang: reject host package: no slang library target")
    return()
endif()
get_target_property(_aliased "${_target}" ALIASED_TARGET)
if(_aliased)
    set(_target "${_aliased}")
endif()
get_target_property(_imported "${_target}" IMPORTED)
if(NOT _imported)
    message(STATUS "Slang: reject host package: expected imported library target")
    return()
endif()
get_target_property(_includes "${_target}" INTERFACE_INCLUDE_DIRECTORIES)
set(_headers_found FALSE)
foreach(_inc IN LISTS _includes)
    foreach(_dir IN ITEMS "${_inc}" "${_inc}/slang" "${_inc}/shader-slang")
        if(EXISTS "${_dir}/slang.h" AND EXISTS "${_dir}/slang-com-helper.h" AND EXISTS "${_dir}/slang-com-ptr.h")
            set(_headers_found TRUE)
        endif()
    endforeach()
endforeach()
if(NOT _headers_found)
    message(STATUS "Slang: reject host package: advertised public headers incomplete")
    return()
endif()
set(_library "")
foreach(_property IN ITEMS IMPORTED_LOCATION IMPORTED_LOCATION_RELEASE IMPORTED_LOCATION_RELWITHDEBINFO IMPORTED_LOCATION_NOCONFIG IMPORTED_LOCATION_NONE)
    get_target_property(_location "${_target}" "${_property}")
    if(_location AND EXISTS "${_location}")
        set(_library "${_location}")
        break()
    endif()
endforeach()
if(NOT _library)
    get_target_property(_configs "${_target}" IMPORTED_CONFIGURATIONS)
    foreach(_config IN LISTS _configs)
        string(TOUPPER "${_config}" _config)
        get_target_property(_location "${_target}" "IMPORTED_LOCATION_${_config}")
        if(_location AND EXISTS "${_location}")
            set(_library "${_location}")
            break()
        endif()
    endforeach()
endif()
if(NOT _library)
    message(STATUS "Slang: reject host package: imported runtime library missing")
    return()
endif()
set(_package_version "")
if(DEFINED slang_VERSION)
    set(_package_version "${slang_VERSION}")
elseif(DEFINED slang_VERSION_STRING)
    set(_package_version "${slang_VERSION_STRING}")
endif()
if(NOT _package_version)
    file(REAL_PATH "${_library}" _real_library)
    if(_real_library MATCHES "([0-9][0-9][0-9][0-9]\\.[0-9]+)")
        set(_package_version "${CMAKE_MATCH_1}")
    endif()
endif()
if(_package_version MATCHES "([0-9][0-9][0-9][0-9]\\.[0-9]+)")
    if(NOT "${CMAKE_MATCH_1}" STREQUAL "${_compiler_version}")
        message(STATUS "Slang: reject host: compiler ${_compiler_version} / library ${_package_version} mismatch")
        return()
    endif()
endif()
if(UNIX)
    find_program(_zahlen_file NAMES file NO_CACHE)
    if(_zahlen_file)
        execute_process(COMMAND "${_zahlen_file}" -Lb "${_library}" RESULT_VARIABLE _rc OUTPUT_VARIABLE _format)
        if(NOT "${_rc}" STREQUAL "0" OR
           (APPLE AND NOT _format MATCHES "Mach-O") OR
           (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT _format MATCHES "ELF") OR
           (CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$" AND NOT _format MATCHES "arm64|aarch64|ARM aarch64") OR
           (CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$" AND NOT _format MATCHES "x86[_-]64|x86-64"))
            message(STATUS "Slang: reject host package: platform/architecture mismatch (${_format})")
            return()
        endif()
    endif()
endif()
# Promote only the accepted library; rejected targets stay directory-local.
set_property(TARGET "${_target}" PROPERTY IMPORTED_GLOBAL TRUE)
set(ZHLN_SLANG_TARGET "${_target}" PARENT_SCOPE)
set(SLANG_EXECUTABLE "${_zahlen_host_slangc}" PARENT_SCOPE)
set(SLANG_COMPILER_DEPENDS "${_zahlen_host_slangc}" PARENT_SCOPE)
message(STATUS "Slang: selected host compiler ${_zahlen_host_slangc}; package ${slang_DIR} (${_compiler_version})")
