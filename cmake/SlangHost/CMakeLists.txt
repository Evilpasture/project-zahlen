# A directory-scoped package probe: rejected imported targets stay in this
# directory and cannot poison the fallback source build in a sibling directory.
# Only libslang is requested -- the shared library and its public headers. No
# version gate, no architecture probe, no compiler executable.
find_package(slang CONFIG QUIET)
if(NOT slang_FOUND)
    message(STATUS "libslang: no host slang config package found")
    return()
endif()
if(TARGET slang::slang)
    set(_target slang::slang)
elseif(TARGET slang)
    set(_target slang)
else()
    message(STATUS "libslang: reject host package: no slang library target")
    return()
endif()
get_target_property(_aliased "${_target}" ALIASED_TARGET)
if(_aliased)
    set(_target "${_aliased}")
endif()
get_target_property(_imported "${_target}" IMPORTED)
if(NOT _imported)
    message(STATUS "libslang: reject host package: expected imported library target")
    return()
endif()
get_target_property(_includes "${_target}" INTERFACE_INCLUDE_DIRECTORIES)
set(_headers_found FALSE)
foreach(_inc IN LISTS _includes)
    set(_dir "${_inc}")
    if(_dir MATCHES "^\\$<(BUILD_INTERFACE|INSTALL_INTERFACE):(.*)>$")
        set(_dir "${CMAKE_MATCH_2}")
    endif()
    foreach(_candidate IN ITEMS "${_dir}" "${_dir}/slang" "${_dir}/shader-slang")
        if(EXISTS "${_candidate}/slang.h" AND
           EXISTS "${_candidate}/slang-com-helper.h" AND
           EXISTS "${_candidate}/slang-com-ptr.h")
            set(_headers_found TRUE)
        endif()
    endforeach()
endforeach()
if(NOT _headers_found)
    message(STATUS "libslang: reject host package: advertised public headers incomplete")
    return()
endif()
set(_library "")
foreach(_property IN ITEMS IMPORTED_LOCATION IMPORTED_LOCATION_RELEASE IMPORTED_LOCATION_RELWITHDEBINFO IMPORTED_LOCATION_NOCONFIG IMPORTED_LOCATION_NONE)
    get_target_property(_location "${_target}" "${_property}")
    if(_location AND EXISTS "${_location}")
        set(_library "${_location}")
        break()
    endif()
endforeach()
if(NOT _library)
    get_target_property(_configs "${_target}" IMPORTED_CONFIGURATIONS)
    foreach(_config IN LISTS _configs)
        string(TOUPPER "${_config}" _config)
        get_target_property(_location "${_target}" "IMPORTED_LOCATION_${_config}")
        if(_location AND EXISTS "${_location}")
            set(_library "${_location}")
            break()
        endif()
    endforeach()
endif()
if(NOT _library)
    message(STATUS "libslang: reject host package: imported runtime library missing")
    return()
endif()
# Promote only the accepted library; rejected targets stay directory-local.
set_property(TARGET "${_target}" PROPERTY IMPORTED_GLOBAL TRUE)
set(ZHLN_SLANG_TARGET "${_target}" PARENT_SCOPE)
message(STATUS "libslang: selected host package (${slang_DIR})")
