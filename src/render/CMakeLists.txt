# ----------------------------------------------------------------------------
# zahlen_render — the renderer
# ----------------------------------------------------------------------------
# RenderContext implementation: frame graph, passes, resource pools, and the
# engine-facing material/mesh factories. Sits on the zahlen_vulkan RHI leaf
# (src/vulkan) and is linked privately into zahlen_engine, same shape as the
# other subsystem targets (zahlen_ecs, zahlen_gui, ...).
#
# This target is the ONE sanctioned consumer of the RHI's implementation
# headers (Rendering.hpp & friends): it carries src/vulkan on its own
# PRIVATE include path instead of inheriting it engine-wide.

set(RAW_RENDER_SOURCES
    RenderFrame.cpp
    RenderDestinations.cpp
    PresentationSurface.cpp
    RenderTexture.cpp
    RenderPresentation.cpp
    DestinationRegistry.cpp
    RenderGraphBuilder.cpp
    RenderDrawCommands.cpp
    RenderInit.cpp
    init/RenderInitDevice.cpp
    init/RenderInitHeaps.cpp
    init/RenderInitTargets.cpp
    init/RenderInitScenePipelines.cpp
    init/RenderInitPostProcess.cpp
    init/RenderInitUI.cpp
    ui/UIRenderer.cpp
    RenderResources.cpp
    MeshBuilder.cpp
    Meshlet.cpp
    TextureManager.cpp
    Resources.cpp
    RenderDocInit.cpp
    RenderProcedural.cpp
    RenderPasses.cpp
    RenderSetup.cpp
    pipelines/ComputeSimPipeline.cpp
    pipelines/DeferredPbrPipeline.cpp
    pipelines/UIPipeline.cpp
)

# Reflection-capable compilers compile these files directly; other compilers
# receive their generated counterparts. Either way, the renderer contributes
# its sources exclusively through this directory.
zahlen_transpile_sources(RENDER_SOURCES ${RAW_RENDER_SOURCES})

# The generated bytecode joins the target after the transpiler was offered the
# hand-written sources: it has no reflection to flatten, and it does not exist
# when this directory is configured.
list(APPEND RENDER_SOURCES "${ZHLN_SHADER_CATALOG_SOURCE}")

add_library(zahlen_render STATIC ${RENDER_SOURCES})
set_target_properties(zahlen_render PROPERTIES POSITION_INDEPENDENT_CODE ON)

target_compile_features(zahlen_render PUBLIC cxx_std_26)
set_target_properties(zahlen_render PROPERTIES
    CXX_STANDARD 26
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
)
if(NOT MSVC)
    target_compile_options(zahlen_render PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions -fno-rtti>)
endif()

target_link_libraries(zahlen_render
    PUBLIC  zahlen_vulkan
    # RenderInternal.hpp speaks JPH math types through the public headers.
    PUBLIC  Jolt
    # No glfw here. The renderer is handed an PresentationTarget and reads the
    # opaque NativeSurfaceHandle behind it; the window system belongs to
    # src/window. What links instead is that subsystem's archive, for the
    # out-of-line half of the seam: every handle and every target the renderer
    # holds is moved and destroyed through members defined over there. The
    # descriptor itself is read by PresentationSurface.cpp through Visit(), which
    # is the handle's one friend -- there is no accessor to reach it another way,
    # and no GLFW type is named from here. The macOS host-blit plugin
    # (OpenGLHacks) is the one remaining GLFW consumer under this directory, and
    # it links its own copy in its own target.
    # The ONE meshoptimizer link in the tree: Meshlet.cpp compiles the
    # BuildMeshlets implementation behind the declaration-only public header,
    # so every consumer (engine, zcook, glTF, terrain, the meshlet tests)
    # resolves the ZHLN_API symbols through libzahlen_engine instead of
    # carrying the include path and the library itself.
    PRIVATE meshoptimizer
    # For the presentation seam's out-of-line PIMPL members (NativeSurfaceHandle
    # and PresentationTarget), not for the window system: glfw is still not
    # linked here.
    PRIVATE zahlen_window
)

# Renderer-private headers, plus the RHI implementation headers this layer is
# allowed to see. Both variants matter: transpiled sources compile from the
# build tree, where quote includes no longer resolve beside the originals.
# src/window is on the path for exactly two headers, PresentationTarget.hpp and
# NativeSurfaceInternal.hpp -- the private half of the presentation seam, and the
# only windowing this layer knows. They are included by bare spelling rather than
# as "window/...", the same way src/vulkan reaches the RHI's own private headers.
target_include_directories(zahlen_render PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/init
    ${PROJECT_SOURCE_DIR}/src/vulkan
    ${PROJECT_SOURCE_DIR}/src/window
)

target_include_directories(zahlen_render SYSTEM PRIVATE
    ${PROJECT_SOURCE_DIR}/extern/stb
    ${PROJECT_SOURCE_DIR}/extern/JoltPhysics
    ${PROJECT_SOURCE_DIR}/third_party/renderdoc
)

# The renderer precompiles the RHI umbrella on GCC, and only there: force
# including it ahead of the P2996 reflection machinery in RenderInternal.hpp
# ICEs clang-p2996 (Expr::isDefaultArgument during splice instantiation), which
# is why the engine disables its PCH on Clang and why this one is guarded by
# compiler rather than by language.
#
# The win is this target's whole header cost. All 26 of its translation units
# open with the same closure -- Vulkan-Headers, VMA, volk and the RHI's own
# headers, 110 headers and ~72,000 lines -- and unlike the engine and the RHI it
# had no PCH to share it: 2.25M lines of headers per full rebuild, 84% of it
# this one file, for the code that changes most. Precompiling src/vulkan/
# Rendering.hpp is exactly the engine's arrangement, and every render TU
# compiles with the umbrella force-included whether or not the PCH is used --
# checked TU by TU before this line was written, because a PCH that silently
# disagrees with the source is a class of bug worth not having.
#
# GCC takes the PCH only where the flags agree and falls back to parsing the
# header where they do not -- this target compiles every TU with the same
# options, so it is used everywhere. Wrong flags would cost time, never
# correctness.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU")
    target_precompile_headers(zahlen_render PRIVATE
        "${PROJECT_SOURCE_DIR}/src/vulkan/Rendering.hpp"
    )
endif()

# The shader catalog, generated by zshader after the cooks (see cmake/
# ShaderCompilation.cmake): ShaderBindings.hpp is what the render sources compile
# against, ShaderBytecode.cpp is the one translation unit that embeds the cooked
# modules. A custom command's outputs are only known in the directory that
# declared them, so they are claimed here as GENERATED.
set_source_files_properties("${ZHLN_SHADER_CATALOG_HEADER}" "${ZHLN_SHADER_CATALOG_SOURCE}" PROPERTIES GENERATED TRUE)
add_dependencies(zahlen_render zahlen_shader_catalog)
target_include_directories(zahlen_render PRIVATE "${CMAKE_BINARY_DIR}/generated_shaders")
# The generated GPU host types (see cmake/ShaderCompilation.cmake): GpuAbi.hpp
# and RenderInternal.hpp compile against GeneratedGpuTypes.hpp, so the
# reflection runs before this target does. Only included, never compiled, so
# unlike the catalog above it needs no GENERATED marking.
add_dependencies(zahlen_render zahlen_gpu_types)

# GpuAbi.hpp holds the engine's generated GPU types against the module that
# declares them, and the GPU ABI's own bytes are what says whether
# they match: `#embed` needs a file name here, and a module's bytes are only a
# constant expression in a translation unit that embedded them -- so the ABI
# module's bytes are embedded by the check itself (see cmake/ShaderCompilation.cmake,
# which keeps them out of the generated bytecode), not handed over at runtime.
target_compile_definitions(zahlen_render PRIVATE ZHLN_GPU_ABI_MODULE="${SHADER_GPU_ABI_CS_PATH}")

target_link_libraries(zahlen_render PRIVATE zahlen_warnings)

add_subdirectory(OpenGLHacks)
